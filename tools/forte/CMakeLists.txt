# FORTE with the exported ModLib and module types, as the FBE's forte recipe builds it.
cmake_minimum_required(VERSION 3.24)
project(forte-local C CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
add_subdirectory(${FORTE_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/forte EXCLUDE_FROM_ALL)
file(GLOB module_dirs "${MODULES_DIR}/*")
foreach (dir IN LISTS module_dirs)
    if (EXISTS ${dir}/CMakeLists.txt)
        message(STATUS "Adding External Module: ${dir}")
        get_filename_component(name ${dir} NAME)
        get_property(bindir_base TARGET forte PROPERTY BINARY_DIR)
        add_subdirectory(${dir} ${bindir_base}/module-${name})
    endif ()
endforeach ()
add_custom_target(forte-all ALL DEPENDS forte)
